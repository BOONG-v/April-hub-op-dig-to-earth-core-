                {name = "🔗 Quick Join Code", value = "```lua\n" .. joinCode .. "\n```", inline = false}
            },
            footer = {text = scriptName .. " • Execution Tracker"},
            timestamp = os.date("!%Y-%m-%dT%H:%M:%SZ")
        }}
    }

    local jsonData = self.services.HttpService:JSONEncode(data)
    local headers = {["Content-Type"] = "application/json"}
    local request = http_request or request or HttpPost or syn.request

    if request then
        pcall(function()
            request({
                Url = self.config.webhookUrl,
                Method = "POST",
                Headers = headers,
                Body = jsonData
            })
        end)
    end
end

function AprilHub:loadPetData()
    if self.placeId ~= "81440632616906" then return true end
    
    local petsModule = self.services.ReplicatedStorage:FindFirstChild("PetsModule")
    if not petsModule then
        warn("PetsModule not found!")
        return false
    end

    local success, petData = pcall(function()
        return require(petsModule)
    end)

    if not success or type(petData) ~= "table" then
        warn("Failed to load pet data!")
        return false
    end

    for petName, info in pairs(petData) do
        if type(petName) == "string" and type(info) == "table" then
            local world = info.World or 1
            if not self.petData.petsByWorld[world] then
                self.petData.petsByWorld[world] = {}
                table.insert(self.petData.allWorlds, world)
            end

            table.insert(self.petData.petsByWorld[world], {
                name = petName,
                multiplier = info.Multiplier or 1,
                rarity = info.Rarity or 1,
                rarityType = info.RarityType or "Common",
                image = info.Image or "",
                world = world
            })
        end
    end

    table.sort(self.petData.allWorlds)
    return true
end

function AprilHub:teleportToWorld(world)
    if self.state.teleportCooldowns[world] and tick() - self.state.teleportCooldowns[world] < 10 then
        self.state.orionLib:MakeNotification({
            Name = "Cooldown",
            Content = "Please wait before teleporting to World " .. world .. " again!",
            Image = "rbxassetid://4483345998",
            Time = 3
        })
        return
    end

    local coords = self.config.gameConfigs[self.placeId].worldCoords[world]
    if coords and self.player.Character and self.player.Character:FindFirstChild("HumanoidRootPart") then
        local success = pcall(function()
            self.player.Character.HumanoidRootPart.CFrame = CFrame.new(coords)
        end)

        if success then
            self.state.teleportCooldowns[world] = tick()
            self.state.orionLib:MakeNotification({
                Name = "Teleported",
                Content = "Teleported to World " .. world .. "!",
                Image = "rbxassetid://4483345998",
                Time = 3
            })
        end
    end
end

function AprilHub:toggleLightShardFarm(enabled)
    if enabled then
        self.state.autoFarmStates.lightShard = true
        self.state.farmingLoops.lightShard = spawn(function()
            while self.state.autoFarmStates.lightShard do
                pcall(function()
                    local args = {"LightShard"}
                    self.services.ReplicatedStorage:WaitForChild("Remotes"):WaitForChild("TreasureEvent"):FireServer(unpack(args))
                end)
                wait(0.2)
            end
        end)
    else
        self.state.autoFarmStates.lightShard = false
        if self.state.farmingLoops.lightShard then
            self.state.farmingLoops.lightShard:Disconnect()
            self.state.farmingLoops.lightShard = nil
        end
    end
end

function AprilHub:toggleAutoPrizeFarm(enabled)
    if enabled then
        self.state.autoFarmStates.autoPrize = true
        self.state.farmingLoops.autoPrize = spawn(function()
            local prizeEvents = {8, 6, 2, 1, 5}
            local currentIndex = 1

            while self.state.autoFarmStates.autoPrize do
                pcall(function()
                    self.services.ReplicatedStorage:WaitForChild("Remotes"):WaitForChild("SpinPrizeEvent"):FireServer(prizeEvents[currentIndex])
                end)

                currentIndex = currentIndex + 1
                if currentIndex > #prizeEvents then
                    currentIndex = 1
                end

                wait(0.5)
            end
        end)
    else
        self.state.autoFarmStates.autoPrize = false
        if self.state.farmingLoops.autoPrize then
            self.state.farmingLoops.autoPrize:Disconnect()
            self.state.farmingLoops.autoPrize = nil
        end
    end
end

function AprilHub:spawnPet(pet, button)
    local petName = pet.name
    local isVoidPet = string.find(petName:lower(), "void ")
    local isGoldPet = string.find(petName:lower(), "gold ")
    local isDiamondPet = string.find(petName:lower(), "diamond ")

    if isVoidPet or isGoldPet or isDiamondPet then
        local basePetName = petName
        if isVoidPet then
            basePetName = petName:gsub("^[Vv]oid ", "")
        elseif isGoldPet then
            basePetName = petName:gsub("^[Gg]old ", "")
        elseif isDiamondPet then
            basePetName = petName:gsub("^[Dd]iamond ", "")
        end

        button.BackgroundColor3 = Color3.fromRGB(255, 193, 7)
        button.Text = "SPAWNING..."

        spawn(function()
            for i = 1, 5 do
                button.Text = "SPAWNING " .. i .. "/5"
                local args = {basePetName}
                self.services.ReplicatedStorage:WaitForChild("Remotes"):WaitForChild("PetCageEvent"):FireServer(unpack(args))
                wait(0.1) 
            end

            button.Text = "CRAFTING..."
            wait(0.5) 

            pcall(function()
                if isVoidPet then
                    game.ReplicatedStorage.PetRemotes.VoidPetCraftEvent:FireServer(basePetName, 100)
                elseif isGoldPet then
                    game.ReplicatedStorage.PetRemotes.GoldPetCraftEvent:FireServer(basePetName, 100)
                elseif isDiamondPet then
                    game.ReplicatedStorage.PetRemotes.DiamondPetCraftEvent:FireServer(basePetName, 100)
                end
            end)

            button.BackgroundColor3 = Color3.fromRGB(25, 135, 84)
            button.Text = "DONE!"

            wait(2)
            button.BackgroundColor3 = Color3.fromRGB(40, 167, 69)
            button.Text = "SPAWN"
        end)
    else
        local success = pcall(function()
            local args = {petName}
            self.services.ReplicatedStorage:WaitForChild("Remotes"):WaitForChild("PetCageEvent"):FireServer(unpack(args))
        end)

        if success then
            button.BackgroundColor3 = Color3.fromRGB(25, 135, 84)
            button.Text = "SPAWNED!"
        else
            button.BackgroundColor3 = Color3.fromRGB(220, 53, 69)
            button.Text = "ERROR!"
        end

        wait(1.5)
        button.BackgroundColor3 = Color3.fromRGB(40, 167, 69)
        button.Text = "SPAWN"
    end
end

function AprilHub:loadDeveloperNames()
    local success, result = pcall(function()
        local response = game:HttpGet("https://raw.githubusercontent.com/l3rrythelob/l3rrythelob/refs/heads/main/names.lua")
        if response then
            local func = loadstring(response)
            if func then
                local names = func() 
                if type(names) == "table" then
                    for i, name in ipairs(names) do
                        self.playerDetection.DEV_USERNAMES[string.lower(name)] = true
                    end
                    return true
                end
            end
        end
        return false
    end)
    
    if not success then
        warn("Failed to load developer names from GitHub:", result)
        self.playerDetection.DEV_USERNAMES["burrylerry"] = true
        self.playerDetection.DEV_USERNAMES["burryterry"] = true
        self.playerDetection.DEV_USERNAMES["burrybeddy"] = true
    end
end

function AprilHub:sendToServer(endpoint, data, callback)
    if self.playerDetection.pendingRequests >= self.playerDetection.maxConcurrentRequests then
        if callback then callback(nil, "Rate limited") end
        return
    end
    
    self.playerDetection.pendingRequests = self.playerDetection.pendingRequests + 1
    
    spawn(function()
        local request = http_request or request or HttpPost or syn.request
        
        if not request then
            self.playerDetection.pendingRequests = self.playerDetection.pendingRequests - 1
            if callback then callback(nil, "No HTTP function") end
            return
        end
        
        local success, result = pcall(function()
            local requestData = {
                Url = self.playerDetection.SERVER_URL .. endpoint,
                Headers = {
                    ["Content-Type"] = "application/json"
                },
                Timeout = self.playerDetection.REQUEST_TIMEOUT
            }
            
            if endpoint == "/api/users" then
                requestData.Method = "GET"
            else
                requestData.Method = "POST"
                requestData.Body = self.services.HttpService:JSONEncode(data)
            end
            
            local response = request(requestData)
            
            if response and response.StatusCode == 200 and response.Body then
                return self.services.HttpService:JSONDecode(response.Body)
            end
            return nil
        end)
        
        self.playerDetection.pendingRequests = self.playerDetection.pendingRequests - 1
        
        if callback then
            if success and result then
                callback(result)
            else
                callback(nil, result or "Request failed")
            end
        end
    end)
end

function AprilHub:getCurrentPlayerList()
    local currentPlayers = {}
    for _, player in pairs(self.services.Players:GetPlayers()) do
        if player ~= self.player then
            table.insert(currentPlayers, player.Name)
        end
    end
    return currentPlayers
end

function AprilHub:findPlayerChanges(oldList, newList)
    local added = {}
    local removed = {}
    
    local oldLookup = {}
    for _, username in pairs(oldList) do
        oldLookup[username] = true
    end
    
    for _, username in pairs(newList) do
        if not oldLookup[username] then
            table.insert(added, username)
        end
    end
    
    local newLookup = {}
    for _, username in pairs(newList) do
        newLookup[username] = true
    end
    
    for _, username in pairs(oldList) do
        if not newLookup[username] then
            table.insert(removed, username)
        end
    end
    
    return added, removed
end

function AprilHub:checkAllCurrentPlayers()
    local allPlayers = self:getCurrentPlayerList()
    
    if #allPlayers == 0 then
        return
    end
    
    for i, username in pairs(allPlayers) do
        spawn(function()
            wait(i * 0.5)
            local player = self.services.Players:FindFirstChild(username)
            if not player or not player.Character or not player.Character:FindFirstChild("Head") then
                return
            end
            
            self:sendToServer("/api/checkUser", {
                username = username,
                placeId = self.placeId
            }, function(response, error)
                if response and response.isActive ~= nil then
                    self.playerDetection.knownPlayers[username] = response.isActive
                    
                    if response.isActive then
                        local displayInfo = self:getUserDisplayInfo(username)
                        self.playerDetection.userDisplayGuis[username] = self:createUserDisplay(player, displayInfo)
                    end
                else
                    self.playerDetection.knownPlayers[username] = false
                end
            end)
        end)
    end
end

function AprilHub:handlePlayerChanges(addedPlayers, removedPlayers)
    for _, username in pairs(removedPlayers) do
        self:removeUserDisplay(username)
        self.playerDetection.knownPlayers[username] = nil
    end
end

function AprilHub:scanForPlayerChanges()
    local currentPlayerList = self:getCurrentPlayerList()
    
    if not self.playerDetection.initialScanComplete then
        self.playerDetection.lastPlayerList = currentPlayerList
        self.playerDetection.initialScanComplete = true
        return
    end
    
    local addedPlayers, removedPlayers = self:findPlayerChanges(self.playerDetection.lastPlayerList, currentPlayerList)
    
    if #addedPlayers > 0 or #removedPlayers > 0 then
        self:handlePlayerChanges(addedPlayers, removedPlayers)
        self.playerDetection.lastPlayerList = currentPlayerList
    end
end

function AprilHub:startPlayerScanning()
    if self.playerDetection.playerScanConnection then
        self.playerDetection.playerScanConnection:Disconnect()
    end
    
    local lastScan = 0
    self.playerDetection.playerScanConnection = self.services.RunService.Heartbeat:Connect(function()
        local currentTime = tick()
        if currentTime - lastScan >= self.playerDetection.PLAYER_SCAN_INTERVAL and self.playerDetection.isConnected then
            lastScan = currentTime
            self:scanForPlayerChanges()
        end
    end)
end

function AprilHub:joinServer()
    local username = self.player.Name
    
    self:sendToServer("/api/join", {
        username = username,
        placeId = self.placeId
    }, function(response, error)
        if response and response.success then
            self.playerDetection.isConnected = true
            
            self.playerDetection.knownPlayers[username] = true
            local displayInfo = self:getUserDisplayInfo(username)
            if self.player.Character and self.player.Character:FindFirstChild("Head") then
                self.playerDetection.userDisplayGuis[username] = self:createUserDisplay(self.player, displayInfo)
            end
            
            if response.usersInPlace then
                for _, user in pairs(response.usersInPlace) do
                    local existingPlayer = self.services.Players:FindFirstChild(user.username)
                    if existingPlayer and existingPlayer ~= self.player then
                        self.playerDetection.knownPlayers[user.username] = true
                        local displayInfo = user.displayInfo or self:getUserDisplayInfo(user.username)
                        self.playerDetection.userDisplayGuis[user.username] = self:createUserDisplay(existingPlayer, displayInfo)
                    end
                end
            end
            
            self:startHeartbeat()
            self:startPlayerScanning()
            self:checkAllCurrentPlayers()
            
        else
            wait(10)
            if not self.playerDetection.isConnected then
                self:joinServer()
            end
        end
    end)
end

function AprilHub:pollServerForNewUsers()
    if not self.playerDetection.isConnected then return end
    
    self:sendToServer("/api/users", {}, function(response, error)
        if response and type(response) == "table" then
            local usersInThisPlace = {}
            
            for _, user in pairs(response) do
                if user.placeId == self.placeId and user.username ~= self.player.Name then
                    usersInThisPlace[user.username] = user
                end
            end
            
            for username, userData in pairs(usersInThisPlace) do
                local player = self.services.Players:FindFirstChild(username)
                if player and not self.playerDetection.knownPlayers[username] then

                    self.playerDetection.knownPlayers[username] = true
                    
                    if player.Character and player.Character:FindFirstChild("Head") then
                        local displayInfo = userData.displayInfo or self:getUserDisplayInfo(username)
                        self.playerDetection.userDisplayGuis[username] = self:createUserDisplay(player, displayInfo)
                        --self:showUserNotification(username, displayInfo, true)
                    end
                end
            end
            
            local currentKnownUsers = {}
            for username, _ in pairs(self.playerDetection.knownPlayers) do
                if username ~= self.player.Name then
                    currentKnownUsers[username] = true
                end
            end
            
            for username, _ in pairs(currentKnownUsers) do
                if not usersInThisPlace[username] then
                    local displayInfo = self:getUserDisplayInfo(username)
                    --self:showUserNotification(username, displayInfo, false)
                    self:removeUserDisplay(username)
                    self.playerDetection.knownPlayers[username] = nil
                end
            end
        end
    end)
end

function AprilHub:startHeartbeat()
    if self.playerDetection.heartbeatConnection then
        self.playerDetection.heartbeatConnection:Disconnect()
    end
    
    local lastHeartbeat = 0
    local lastUserPoll = 0
    local HEARTBEAT_INTERVAL = 10 
    local USER_POLL_INTERVAL = 12 
    
    self.playerDetection.heartbeatConnection = self.services.RunService.Heartbeat:Connect(function()
        local currentTime = tick()
        
        if currentTime - lastHeartbeat >= HEARTBEAT_INTERVAL then
            lastHeartbeat = currentTime
            
            self:sendToServer("/api/heartbeat", {
                username = self.player.Name,
                placeId = self.placeId
            })
        end
        
        if currentTime - lastUserPoll >= USER_POLL_INTERVAL then
            lastUserPoll = currentTime
            self:pollServerForNewUsers()
        end
    end)
end

function AprilHub:setupDisconnectDetection()

    
    
    self.services.Players.PlayerRemoving:Connect(function(player)
        if player == self.player then
            self:immediateDisconnect()
        end
    end)

    self.player.CharacterRemoving:Connect(function()
        spawn(function()
            wait(3) 
            if not self.player.Character then
                self:immediateDisconnect()
            end
        end)
    end)
end

function AprilHub:immediateDisconnect()
    if self.playerDetection.cleanupExecuted then return end
    
    spawn(function()
        local httpRequest = http_request or request or HttpPost or syn.request
        if httpRequest and self.player then
            pcall(function()
                httpRequest({
                    Url = self.playerDetection.SERVER_URL .. "/api/disconnect",
                    Method = "POST",
                    Headers = {
                        ["Content-Type"] = "application/json"
                    },
                    Body = self.services.HttpService:JSONEncode({
                        username = self.player.Name,
                        placeId = self.placeId
                    }),
                    Timeout = 3
                })
            end)
        end
    end)
    
    self:cleanupPlayerDetection()
end

function AprilHub:showUserNotification(username, displayInfo, isJoining)
    if not self.state.orionLib then return end
    
    local action = isJoining and "joined" or "left"
    local message = username .. " " .. action .. " with " .. displayInfo.displayName
    
    self.state.orionLib:MakeNotification({
        Name = isJoining and "User Joined" or "User Left",
        Content = message,
        Image = "rbxassetid://4483345998",
        Time = 4
    })
end

function AprilHub:cleanupPlayerDetection()
    if self.playerDetection.cleanupExecuted then
        return
    end
    self.playerDetection.cleanupExecuted = true
    
    if self.playerDetection.heartbeatConnection then
        self.playerDetection.heartbeatConnection:Disconnect()
        self.playerDetection.heartbeatConnection = nil
    end
    
    if self.playerDetection.playerScanConnection then
        self.playerDetection.playerScanConnection:Disconnect()
        self.playerDetection.playerScanConnection = nil
    end
    
    for _, gui in pairs(self.playerDetection.userDisplayGuis) do
        if gui and gui.Parent then
            gui:Destroy()
        end
    end
    self.playerDetection.userDisplayGuis = {}
    
    self.playerDetection.isConnected = false
    
    spawn(function()
        local httpRequest = http_request or _G.request or HttpPost or syn and syn.request
        if httpRequest and self.player then
            pcall(function()
                httpRequest({
                    Url = self.playerDetection.SERVER_URL .. "/api/disconnect",
                    Method = "POST",
                    Headers = {
                        ["Content-Type"] = "application/json"
                    },
                    Body = self.services.HttpService:JSONEncode({
                        username = self.player.Name,
                        placeId = self.placeId
                    }),
                    Timeout = 5
                })
            end)
        end
    end)
end

function AprilHub:initializePlayerDetection()
    self:loadDeveloperNames()
    
    local character = self.player.Character or self.player.CharacterAdded:Wait()
    if not character:FindFirstChild("Head") then
        character:WaitForChild("Head")
    end
    
    self:joinServer()
end

function AprilHub:initialize()
    task.wait(1)
    self:sendWebhookMessage()

    if self.placeId == "81440632616906" then
        if not self:loadPetData() then
            print("Failed to load pet data!")
            return false
        end
    end

    self:createOrionGUI()
    
    self:initializePlayerDetection()
    
    self:setupDisconnectDetection()
    
    local cleanupConnection
    cleanupConnection = self.services.RunService.Heartbeat:Connect(function()
        if not self.player or not self.player.Parent or not workspace:FindFirstChild(self.player.Name) then
            self:cleanupPlayerDetection()
            cleanupConnection:Disconnect()
        end
    end)
    
    local gameName = self.config.gameConfigs[self.placeId].name
    print("April Hub " .. self.config.version .. " - " .. gameName .. " loaded successfully!")
    return true
end
